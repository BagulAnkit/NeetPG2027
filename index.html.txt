    function updateOngoingTaskBanner() {
      const now = new Date();
      const currentH = now.getHours();
      const currentM = now.getMinutes();
      const currentMinutesFromMidnight = currentH * 60 + currentM;

      const todayTasks = appState.tasks.filter(t => t.date === getTodayYMD() && !t.completed);
      let ongoing = null;

      for (const t of todayTasks) {
        if (t.time) {
          const [th, tm] = t.time.split(':').map(Number);
          const startM = th * 60 + tm;
          const endM = startM + (t.duration || 60);
          if (currentMinutesFromMidnight >= startM && currentMinutesFromMidnight <= endM) {
            ongoing = t;
            break;
          }
        }
      }

      if (ongoing) {
        document.getElementById('ongoingTaskText').textContent = `${ongoing.name} (${ongoing.time})`;
      } else if (todayTasks.length > 0) {
        document.getElementById('ongoingTaskText').textContent = `Next: ${todayTasks[0].name} (${todayTasks[0].duration}m)`;
      } else {
        document.getElementById('ongoingTaskText').textContent = 'All scheduled tasks completed for today!';
      }
    }

    let timerInterval = null;
    let timerTotalSeconds = 40 * 60; // default 40 min
    let timerRemainingSeconds = 40 * 60;
    let timerRunning = false;
    let timerStartTimestamp = null;
    let focusType = 'Study';
    let isPomodoroActive = false;

      focusType = type;

    function setTimerPreset(mins) {
      if (timerRunning) return;
      timerTotalSeconds = mins * 60;
      timerRemainingSeconds = timerTotalSeconds;
      updateTimerDisplay();
    }

    function setCustomTimerDuration() {
      if (timerRunning) return;
      const mins = prompt('Enter countdown minutes:', '45');
      if (mins && !isNaN(mins)) {

    function togglePomodoroMode() {
      isPomodoroActive = !isPomodoroActive;
      document.getElementById('pomoToggleBtn').classList.toggle('bg-amber-500', isPomodoroActive);
      document.getElementById('pomoToggleBtn').classList.toggle('text-white', isPomodoroActive);
      setTimerPreset(isPomodoroActive ? 25 : 40);
    }

    function toggleFocusTimer() {
      if (timerRunning) {
        // Pause
        timerRemainingSeconds = getRemainingSecondsPrecise();
        clearInterval(timerInterval);
        timerRunning = false;
        document.getElementById('focusPlayText').textContent = 'Resume';
        document.getElementById('focusPlayIcon').setAttribute('data-lucide', 'play');
      } else {
        // Start Reverse Countdown
        timerRunning = true;
        timerStartTimestamp = Date.now();
        document.getElementById('focusPlayText').textContent = 'Pause';
        document.getElementById('focusPlayIcon').setAttribute('data-lucide', 'pause');

        timerInterval = setInterval(() => {
          const current = getRemainingSecondsPrecise();
          if (current <= 0) {
            clearInterval(timerInterval);
            timerRunning = false;
            finishAndSaveSession();
            try { new Audio("https://actions.google.com/sounds/v1/alarms/beep_short.ogg").play(); } catch(e){}
          }
          updateTimerDisplay();
        }, 500);
      }
      lucide.createIcons();
    }

    function getRemainingSecondsPrecise() {
      if (!timerRunning) return timerRemainingSeconds;
      const elapsed = Math.floor((Date.now() - timerStartTimestamp) / 1000);
      return Math.max(0, timerRemainingSeconds - elapsed);
    }

    function resetFocusTimer() {
      clearInterval(timerInterval);
      timerRunning = false;
      timerRemainingSeconds = timerTotalSeconds;
      updateTimerDisplay();
      document.getElementById('focusPlayText').textContent = 'Start';
      document.getElementById('focusPlayIcon').setAttribute('data-lucide', 'play');
      lucide.createIcons();
    }

    function updateTimerDisplay() {
      const s = getRemainingSecondsPrecise();
      const h = Math.floor(s / 3600);
      const m = Math.floor((s % 3600) / 60);
      const remSec = s % 60;
      document.getElementById('focusTimerDisplay').textContent =
        `${h < 10 ? '0' : ''}${h}:${m < 10 ? '0' : ''}${m}:${remSec < 10 ? '0' : ''}${remSec}`;
    }

    function finishAndSaveSession() {
      const elapsed = timerTotalSeconds - getRemainingSecondsPrecise();
      if (elapsed < 30) {
        alert('Session too short to record (< 30 seconds).');
        return;
      }

      const subId = document.getElementById('focusSubjectSelect').value;
      const chId = document.getElementById('focusChapterSelect').value;

      appState.focusSessions.unshift({
        id: 'foc_' + Date.now(),
        date: getTodayYMD(),
        type: focusType === 'Other' ? appState.otherTypeName : focusType,
        subjectId: subId,
        chapterId: chId,
        durationSeconds: elapsed,
        timestamp: Date.now()
      });

      // Update Chapter DB
      const sub = appState.subjects.find(s => s.id === subId);
      if (sub) {
        const ch = sub.chapters.find(c => c.id === chId);
        if (ch) {
          ch.totalFocusSeconds = (ch.totalFocusSeconds || 0) + elapsed;
          ch.lastStudied = getTodayYMD();
          if (focusType === 'Revision') ch.status = 'DONE';
        }
      }

      saveState();
      resetFocusTimer();
      alert('Focus session saved and integrated across app!');
      refreshAllViews();
    }

    function renderMonthlyShadedGrid() {
      const grid = document.getElementById('monthlyGridCalendar');
      grid.innerHTML = '';
      const now = new Date();
      const year = now.getFullYear();
      const month = now.getMonth();

      const firstDay = new Date(year, month, 1).getDay();
      const totalDays = new Date(year, month + 1, 0).getDate();

      // Day headers
      ['S', 'M', 'T', 'W', 'T', 'F', 'S'].forEach(h => {
        grid.innerHTML += `<span class="font-bold text-[10px] text-themeMuted">${h}</span>`;
      });

      // Blank prefix
      for (let i = 0; i < firstDay; i++) {
        grid.innerHTML += `<span></span>`;
      }

      // Fill days with shading logic
      for (let day = 1; day <= totalDays; day++) {
        const ymd = `${year}-${String(month + 1).padStart(2, '0')}-${String(day).padStart(2, '0')}`;
        const sec = appState.focusSessions.filter(s => s.date === ymd).reduce((a, b) => a + b.durationSeconds, 0);
        const hours = sec / 3600;

        let bgClass = 'bg-themeCardSoft/40 text-themeMuted'; // <3h
        if (hours >= 4) {
          bgClass = 'bg-purple-600 text-white font-black'; // >4h completed
        } else if (hours >= 2) {
          bgClass = 'bg-purple-400/30 text-purple-300 font-bold'; // light
        }

        grid.innerHTML += `
          <div class="h-7 rounded-lg flex items-center justify-center text-[10px] ${bgClass}">
            ${day}
          </div>
        `;
      }
    }

      queueContainer.innerHTML = '';
      histContainer.innerHTML = '';

      let revList = [];
      appState.subjects.forEach(s => {
        s.chapters.filter(c => c.status === 'R').forEach(ch => {

        });
      });

      if (revList.length === 0) {
        queueContainer.innerHTML = `<p class="text-xs text-themeMuted py-4 text-center">🎉 No chapters flagged for revision!</p>`;
      } else {
        revList.forEach(({ sub, ch }) => {
          const div = document.createElement('div');
          div.className = 'p-3 rounded-2xl bg-themeCard border border-themeBorder flex items-center justify-between';
          div.innerHTML = `
            <div>
              <span class="text-[10px] font-bold uppercase text-amber-400">${sub.name}</span>
              <h5 class="text-xs font-bold text-themeText">${ch.name}</h5>
              <p class="text-[10px] text-themeMuted">Last Studied: ${ch.lastStudied || 'Never'}</p>
            </div>
            <button onclick="markRevisionCompleted('${sub.id}', '${ch.id}')" class="px-3 py-1.5 rounded-xl bg-emerald-500 text-white font-bold text-xs flex items-center gap-1 shadow">
              <i data-lucide="check" class="w-3.5 h-3.5"></i> Mark Done

        appState.revisionHistory.forEach((rh, idx) => {

              <p class="text-[10px] text-themeMuted">${rh.date}</p>
            </div>
            <button onclick="undoRevisionCompletion(${idx})" class="text-[10px] font-bold text-rose-400 underline">Undo</button>

          histContainer.appendChild(div);
        });

    function markRevisionCompleted(subId, chId) {
      const sub = appState.subjects.find(s => s.id === subId);
      const ch = sub?.chapters.find(c => c.id === chId);
      if (ch) {
        ch.status = 'DONE';
        appState.revisionHistory.unshift({
          subId, chId, subName: sub.name, chName: ch.name, date: getTodayYMD()
        });
        saveState();
        renderRevisionTab();
        renderSyllabus();
      }
    }

    function undoRevisionCompletion(index) {
      const item = appState.revisionHistory[index];
      if (item) {
        const sub = appState.subjects.find(s => s.id === item.subId);
        const ch = sub?.chapters.find(c => c.id === item.chId);
        if (ch) ch.status = 'R';
        appState.revisionHistory.splice(index, 1);
        saveState();
        renderRevisionTab();
        renderSyllabus();
      }
    }

          <div class="flex items-center gap-2 pb-1 border-b border-themeBorder">
            <span class="text-xs font-black px-2 py-0.5 rounded bg-themeCardSoft" style="color:${sub.color}">${sub.symbol}</span>
            <h4 class="text-xs font-black uppercase text-themeText">${sub.name}</h4>
          </div>
          <div class="space-y-1.5" id="mistakesList-${sub.id}"></div>
        `;
        container.appendChild(subCard);

        const list = subCard.querySelector(`#mistakesList-${sub.id}`);
        subMistakes.forEach(m => {
          const row = document.createElement('div');
          row.className = 'p-2.5 rounded-xl bg-themeCardSoft border border-themeBorder flex items-center justify-between gap-2 text-xs';
          row.innerHTML = `
            <div class="flex items-center gap-2 truncate flex-1">
              <button onclick="toggleMistakeRectified('${m.id}')" class="w-4 h-4 rounded border flex items-center justify-center ${m.status === 'Corrected' ? 'bg-emerald-500 border-emerald-500 text-white' : 'border-themeMuted'}">
                ${m.status === 'Corrected' ? '<i data-lucide="check" class="w-3 h-3 stroke-[3]"></i>' : ''}
              </button>
              <div class="truncate">
                <span class="font-bold text-themeText block truncate ${m.status === 'Corrected' ? 'line-through text-themeMuted' : ''}">${m.topic}</span>
                ${m.notes ? `<p class="text-[10px] text-themeMuted truncate">${m.notes}</p>` : ''}
              </div>
            </div>
            <button onclick="addMistakeToTaskPrompt('${m.id}')" class="text-[10px] font-bold px-2 py-1 rounded-lg bg-themeCard border border-themeBorder text-themeAccent shrink-0">
              Add to Task?
            </button>
          `;
          list.appendChild(row);
        });
      });
      lucide.createIcons();
    }

    function toggleMistakeRectified(id) {
      const m = appState.mistakes.find(x => x.id === id);
      if (m) {
        m.status = m.status === 'Corrected' ? 'Unresolved' : 'Corrected';
        saveState();
        renderMistakesTracker();
      }
    }

    function addMistakeToTaskPrompt(id) {
      const m = appState.mistakes.find(x => x.id === id);
      if (!m) return;
      const date = prompt('Schedule date (YYYY-MM-DD):', getTodayYMD());
      const time = prompt('Schedule time (e.g. 10:00):', '10:00');
      const dur = prompt('Duration in minutes:', '30');

      if (date) {
        appState.tasks.push({
          id: 'tsk_' + Date.now(),
          name: `Analyse Mistake: ${m.topic}`,
          date,
          time: time || '10:00',
          duration: parseInt(dur) || 30,
          completed: false,
          subjectId: m.subjectId,
          chapterId: m.chapterId
        });
        saveState();
        renderHome();
        alert('Added to schedule tasks!');
      }
    }

    function saveMistake() {
      const topic = document.getElementById('mistakeTopic').value.trim();
      if (!topic) return;
      appState.mistakes.unshift({
        id: 'mst_' + Date.now(),
        subjectId: document.getElementById('mistakeSubject').value,
        chapterId: document.getElementById('mistakeChapter').value,
        topic,
        notes: document.getElementById('mistakeNotes').value.trim(),
        status: 'Unresolved',
        date: getTodayYMD()
      });
      document.getElementById('mistakeTopic').value = '';
      document.getElementById('mistakeNotes').value = '';
      saveState();
      closeModal('mistakeModal');
      renderMistakesTracker();
    }

    // Gym Tracker Controller
    function renderGymTracker() {
      const table = document.getElementById('gymLogTable');
      table.innerHTML = '';
      if (appState.gymLogs.length === 0) {
        table.innerHTML = `<p class="text-center text-xs text-themeMuted py-4">No workouts logged yet.</p>`;
        return;
      }
      appState.gymLogs.forEach(g => {
        table.innerHTML += `
          <div class="p-3 rounded-2xl bg-themeCard border border-themeBorder flex justify-between items-center text-xs">
            <div>
              <span class="font-bold text-themeText">${g.date}</span>
              <p class="text-[11px] text-themeMuted">${g.split}</p>
            </div>
            ${g.weight ? `<span class="text-themeAccent font-bold text-[11px]">${g.weight} kg</span>` : ''}
          </div>
        `;
      });
    }

    function saveGymLog() {
      const split = document.getElementById('gymSplitInput').value.trim();
      const date = document.getElementById('gymDateInput').value || getTodayYMD();
      const weight = parseFloat(document.getElementById('gymWeightInput').value) || null;
      if (!split) return;
      appState.gymLogs.unshift({ date, split, weight });
      saveState();
      closeModal('gymModal');
      renderGymTracker();
    }

    // Nutrition Controller
    function renderNutritionTracker() {
      const container = document.getElementById('nutritionTableContainer');
      container.innerHTML = '';
      if (appState.nutritionLogs.length === 0) {
        container.innerHTML = `<p class="text-center text-xs text-themeMuted py-4">No nutrition entries logged.</p>`;
        return;
      }

      let totalP = 0;
      let totalKcal = 0;
      appState.nutritionLogs.forEach(n => {
        totalP += n.protein;
        totalKcal += n.kcal;
        container.innerHTML += `
          <div class="p-3 rounded-2xl bg-themeCard border border-themeBorder flex justify-between items-center text-xs">
            <span class="font-bold text-themeText">${n.date}</span>
            <div class="flex gap-3">
              <span class="text-themeAccent font-bold">${n.protein}g Protein</span>
              <span class="text-themeMuted">${n.kcal} kcal</span>
            </div>
          </div>
        `;
      });

      document.getElementById('nutritionTotalsDisplay').textContent = `Total: ${totalP}g protein • ${totalKcal} kcal`;
      const avgP = Math.round(totalP / appState.nutritionLogs.length);
      const avgC = Math.round(totalKcal / appState.nutritionLogs.length);
      document.getElementById('nutritionAveragesDisplay').textContent = `Tracked Days: ${appState.nutritionLogs.length} | Avg: ${avgP}g / ${avgC} kcal`;
    }

    function saveNutritionLog() {
      const date = document.getElementById('nutritionDateInput').value || getTodayYMD();
      const protein = parseInt(document.getElementById('nutritionProteinInput').value) || 0;
      const kcal = parseInt(document.getElementById('nutritionKcalInput').value) || 0;
      appState.nutritionLogs.unshift({ date, protein, kcal });
      saveState();
      closeModal('nutritionModal');
      renderNutritionTracker();
    }

    // GT Hub Controller
    function renderGTTracker() {
      const container = document.getElementById('gtListContainer');
      container.innerHTML = '';
      if (appState.grandTests.length === 0) {
        container.innerHTML = `<p class="text-center text-xs text-themeMuted py-6">No Grand Tests recorded yet.</p>`;

                <p class="text-[10px] text-themeMuted">${gt.platform} • ${gt.date}</p>

              <div class="text-right">
                <span class="text-base font-black text-themeAccent">${gt.score} / ${gt.total}</span>
                <p class="text-[10px] font-bold text-emerald-400">${pct}% accuracy</p>
              </div>
            </div>
          </div>
        `;
      });
    }

    function saveGT() {

      });

      renderHome();
    }

    // STATISTICS CONTROLLER
    // ========================================================
    function renderStatistics() {
      const totalSec = appState.focusSessions.reduce((a, b) => a + b.durationSeconds, 0);
      document.getElementById('statsTotalStudyHours').textContent = `${(totalSec / 3600).toFixed(1)}h`;

      let totalMCQs = appState.mcqCountExtra;
      appState.grandTests.forEach(gt => totalMCQs += (gt.attempted || gt.total || 0));
      document.getElementById('statsTotalMCQs').textContent = totalMCQs;

      document.getElementById('statsUnresolvedMistakes').textContent = appState.mistakes.filter(m => m.status !== 'Corrected').length;

      if (appState.grandTests.length > 0) {
        const totalScore = appState.grandTests.reduce((a, b) => a + b.score, 0);
        const totalMarks = appState.grandTests.reduce((a, b) => a + b.total, 0);
        document.getElementById('statsAvgGTScore').textContent = `${Math.round((totalScore / totalMarks) * 100)}%`;
      } else {
        document.getElementById('statsAvgGTScore').textContent = '0%';
      }

      // Subject-wise hours and chapters done
      const list = document.getElementById('subjectWiseStatsList');
      list.innerHTML = '';
      appState.subjects.forEach(s => {
        const sec = appState.focusSessions.filter(fs => fs.subjectId === s.id).reduce((a, b) => a + b.durationSeconds, 0);
        const hrs = (sec / 3600).toFixed(1);
        const doneCh = s.chapters.filter(c => c.status === 'DONE').length;

        list.innerHTML += `
          <div class="p-3 rounded-2xl bg-themeCardSoft border border-themeBorder flex items-center justify-between text-xs">
            <div class="flex items-center gap-2">
              <span class="text-xs font-black px-1.5 py-0.5 rounded" style="background-color:${s.color}20; color:${s.color}">${s.symbol}</span>
              <span class="font-bold text-themeText">${s.name}</span>
            </div>
            <div class="text-right">
              <span class="font-black text-themeAccent">${hrs} hours</span>
              <p class="text-[10px] text-themeMuted">${doneCh}/${s.chapters.length} chapters done</p>
            </div>
          </div>
        `;
      });
    }

    function saveNewSubject() {
      const name = document.getElementById('newSubjectName').value.trim();
      const color = document.getElementById('newSubjectColor').value;

      const symbol = name.substring(0, 2).toUpperCase();
      appState.subjects.push({
        id: 'sub_' + Date.now(),
        name,
        symbol,

        chapters: []
      });
      document.getElementById('newSubjectName').value = '';
      saveState();

    }

    function saveNewChapter() {
      const subId = document.getElementById('addChapterSubjectSelect').value;
      const name = document.getElementById('newChapterName').value.trim();
      const notes = document.getElementById('newChapterNotes').value.trim();
      if (!name) return;
      const sub = appState.subjects.find(s => s.id === subId);
      if (sub) {
        sub.chapters.push({
          id: 'ch_' + Date.now(),
          name,
          notes,
          status: 'BLANK',
          totalFocusSeconds: 0,
          lastStudied: null
        });
        saveState();
        document.getElementById('newChapterName').value = '';
        document.getElementById('newChapterNotes').value = '';
        closeModal('addChapterModal');
        renderSyllabus();
      }
    }

    function exportDataBackup() {
      const str = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(appState, null, 2));
      const a = document.createElement('a');
      a.setAttribute("href", str);
      a.setAttribute("download", `TRACKEBILITY_BACKUP_${getTodayYMD()}.json`);
      document.body.appendChild(a);
      a.click();
      a.remove();
    }

    function importDataBackup(e) {
      const file = e.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = function(evt) {
        try {
          appState = JSON.parse(evt.target.result);
          saveState();
          alert('Database restored successfully!');
          window.location.reload();
        } catch(err) { alert('Invalid backup file format.'); }
      };
      reader.readAsText(file);
    }
